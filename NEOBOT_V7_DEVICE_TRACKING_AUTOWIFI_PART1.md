```kotlin
// backend/services/device-tracking/android/DeviceTrackingService.kt

package com.neobot.tracking

import android.app.Service
import android.content.Intent
import android.location.Location
import android.location.LocationListener
import android.location.LocationManager
import android.os.IBinder
import kotlinx.coroutines.*

class DeviceTrackingService : Service(), LocationListener {
    
    private lateinit var locationManager: LocationManager
    private val serviceScope = CoroutineScope(Dispatchers.IO + SupervisorJob())
    
    override fun onCreate() {
        super.onCreate()
        initializeTracking()
        startForegroundService()
    }
    
    private fun initializeTracking() {
        locationManager = getSystemService(LOCATION_SERVICE) as LocationManager
        
        // GPS tracking
        locationManager.requestLocationUpdates(
            LocationManager.GPS_PROVIDER,
            30000L, // 30 seconds
            10f,    // 10 meters
            this
        )
        
        // Network tracking (WiFi + Cell Tower)
        locationManager.requestLocationUpdates(
            LocationManager.NETWORK_PROVIDER,
            60000L, // 1 minute
            50f,    // 50 meters
            this
        )
        
        // Start monitoring services
        startMonitoring()
    }
    
    private fun startMonitoring() {
        serviceScope.launch {
            // Monitor SIM changes
            monitorSIMChanges()
            
            // Monitor network changes
            monitorNetworkChanges()
            
            // Monitor device usage
            monitorDeviceUsage()
            
            // Periodic photo capture
            capturePeriodicPhotos()
            
            // Backup important data
            backupData()
        }
    }
    
    override fun onLocationChanged(location: Location) {
        serviceScope.launch {
            // Send location to server
            sendLocationUpdate(location)
            
            // Check geofencing
            checkGeofences(location)
            
            // Update local cache
            cacheLocation(location)
        }
    }
    
    private suspend fun sendLocationUpdate(location: Location) {
        val locationData = LocationData(
            latitude = location.latitude,
            longitude = location.longitude,
            accuracy = location.accuracy,
            provider = location.provider,
            timestamp = System.currentTimeMillis(),
            battery = getBatteryLevel(),
            networkInfo = getNetworkInfo()
        )
        
        // Send to Neobot server
        NeobotAPI.sendLocation(locationData)
    }
    
    private suspend fun monitorSIMChanges() {
        val telephonyManager = getSystemService(TELEPHONY_SERVICE) as TelephonyManager
        val currentSIM = telephonyManager.simSerialNumber
        
        if (currentSIM != getStoredSIM()) {
            // SIM changed - ALERT!
            handleSIMChange(currentSIM)
        }
    }
    
    private suspend fun handleSIMChange(newSIM: String) {
        // 1. Capture front camera photo
        val photo = capturePhoto(CAMERA_FRONT)
        
        // 2. Get current location
        val location = getCurrentLocation()
        
        // 3. Send alert to owner
        NeobotAPI.sendAlert(
            type = "SIM_CHANGED",
            data = mapOf(
                "old_sim" to getStoredSIM(),
                "new_sim" to newSIM,
                "location" to location,
                "photo" to photo,
                "timestamp" to System.currentTimeMillis()
            )
        )
        
        // 4. Enable stealth mode
        enableStealthMode()
    }
    
    private suspend fun capturePhoto(camera: Int): ByteArray {
        // Silent photo capture
        val cameraManager = CameraManager()
        return cameraManager.capturePhoto(
            camera = camera,
            silent = true,
            quality = 80
        )
    }
    
    private suspend fun monitorNetworkChanges() {
        val connectivityManager = getSystemService(CONNECTIVITY_SERVICE) as ConnectivityManager
        
        connectivityManager.registerDefaultNetworkCallback(object : ConnectivityManager.NetworkCallback() {
            override fun onAvailable(network: Network) {
                // WiFi connected
                val wifiInfo = getWiFiInfo()
                NeobotAPI.logNetworkChange("connected", wifiInfo)
            }
            
            override fun onLost(network: Network) {
                // WiFi disconnected
                NeobotAPI.logNetworkChange("disconnected", null)
            }
        })
    }
    
    private fun getWiFiInfo(): WiFiInfo {
        val wifiManager = applicationContext.getSystemService(WIFI_SERVICE) as WifiManager
        val connectionInfo = wifiManager.connectionInfo
        
        return WiFiInfo(
            ssid = connectionInfo.ssid,
            bssid = connectionInfo.bssid,
            ipAddress = connectionInfo.ipAddress,
            linkSpeed = connectionInfo.linkSpeed,
            rssi = connectionInfo.rssi
        )
    }
    
    private suspend fun checkGeofences(location: Location) {
        val geofences = getGeofences()
        
        for (geofence in geofences) {
            val distance = calculateDistance(
                location.latitude, location.longitude,
                geofence.latitude, geofence.longitude
            )
            
            if (distance > geofence.radius) {
                // Outside safe zone!
                NeobotAPI.sendAlert(
                    type = "GEOFENCE_EXIT",
                    data = mapOf(
                        "zone_name" to geofence.name,
                        "distance" to distance,
                        "location" to location
                    )
                )
            }
        }
    }
    
    override fun onBind(intent: Intent?): IBinder? = null
}

// Data models
```

### **B. Auto WiFi Connection Service (Android)**

```kotlin
// backend/services/auto-wifi/android/AutoWiFiService.kt

package com.neobot.autowifi

import android.app.Service
import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.content.IntentFilter
import android.net.ConnectivityManager
import android.net.Network
import android.net.NetworkCapabilities
import android.net.wifi.WifiConfiguration
import android.net.wifi.WifiManager
import android.net.wifi.WifiNetworkSpecifier
import android.os.IBinder
import kotlinx.coroutines.*

class AutoWiFiService : Service() {
    
    private lateinit var wifiManager: WifiManager
    private lateinit var connectivityManager: ConnectivityManager
    private val serviceScope = CoroutineScope(Dispatchers.IO + SupervisorJob())
    
    private var isInternetAvailable = true
    private var scanJob: Job? = null
    
    override fun onCreate() {
        super.onCreate()
        
        wifiManager = applicationContext.getSystemService(WIFI_SERVICE) as WifiManager
        connectivityManager = getSystemService(CONNECTIVITY_SERVICE) as ConnectivityManager
        
        startMonitoring()
    }
    
    private fun startMonitoring() {
        // Monitor internet connectivity
        monitorInternetConnection()
        
        // Register WiFi scan receiver
        registerWiFiScanReceiver()
        
        // Start periodic scanning
        startPeriodicScanning()
    }
    
    private fun monitorInternetConnection() {
        connectivityManager.registerDefaultNetworkCallback(object : ConnectivityManager.NetworkCallback() {
            
            override fun onAvailable(network: Network) {
                isInternetAvailable = true
                log("Internet available")
            }
            
            override fun onLost(network: Network) {
                isInternetAvailable = false
                log("Internet lost - triggering auto WiFi")
                
                // Internet down! Auto-connect to WiFi
                serviceScope.launch {
                    autoConnectToWiFi()
                }
            }
            
            override fun onCapabilitiesChanged(
                network: Network,
                capabilities: NetworkCapabilities
            ) {
                val hasInternet = capabilities.hasCapability(
                    NetworkCapabilities.NET_CAPABILITY_INTERNET
                ) && capabilities.hasCapability(
                    NetworkCapabilities.NET_CAPABILITY_VALIDATED
                )
                
                if (!hasInternet && isInternetAvailable) {
                    // Internet quality degraded
                    isInternetAvailable = false
                    serviceScope.launch {
                        autoConnectToWiFi()
                    }
                }
            }
        })
    }
    
    private fun startPeriodicScanning() {
        scanJob = serviceScope.launch {
            while (isActive) {
                if (!isInternetAvailable) {
                    scanAndConnectWiFi()
                }
                delay(30000) // Scan every 30 seconds
            }
        }
    }
    
    private suspend fun autoConnectToWiFi() {
        log("Starting auto WiFi connection...")
        
        // 1. Scan available networks
        val networks = scanAvailableNetworks()
        
        // 2. Rank networks
        val rankedNetworks = rankNetworks(networks)
        
        // 3. Try to connect to best network
        for (network in rankedNetworks) {
            if (tryConnectToNetwork(network)) {
                log("Successfully connected to: ${network.ssid}")
                break
            }
        }
    }
    
    private suspend fun scanAvailableNetworks(): List<WiFiNetwork> {
        wifiManager.startScan()
        
        // Wait for scan results
        delay(5000)
        
        val scanResults = wifiManager.scanResults
        return scanResults.map { result ->
            WiFiNetwork(
                ssid = result.SSID,
                bssid = result.BSSID,
                level = result.level,
                frequency = result.frequency,
                capabilities = result.capabilities,
                isKnown = isKnownNetwork(result.SSID)
            )
        }
    }
    
    private fun rankNetworks(networks: List<WiFiNetwork>): List<WiFiNetwork> {
        return networks
            .filter { it.isKnown } // Only known networks
            .filter { !isBlacklisted(it.ssid) } // Not blacklisted
            .sortedByDescending { calculateScore(it) }
    }
    
    private fun calculateScore(network: WiFiNetwork): Double {
        var score = 0.0
        
        // Signal strength (40%)
        val signalScore = (network.level + 100) / 100.0 // Normalize to 0-1
        score += signalScore * 0.4
        
        // Historical speed (30%)
        val speedScore = getHistoricalSpeed(network.ssid) / 100.0 // Mbps normalized
        score += speedScore * 0.3
        
        // Security (20%)
        val securityScore = if (network.capabilities.contains("WPA")) 1.0 else 0.5
        score += securityScore * 0.2
        
        // Reliability (10%)
        val reliabilityScore = getReliabilityScore(network.ssid)
        score += reliabilityScore * 0.1
        
        return score
    }
    
    private suspend fun tryConnectToNetwork(network: WiFiNetwork): Boolean {
        // Get stored credentials
        val credentials = getStoredCredentials(network.ssid) ?: return false
        
        // Build WiFi configuration
        val wifiConfig = WifiConfiguration().apply {
            SSID = "\"${network.ssid}\""
            
            when {
                network.capabilities.contains("WPA2") -> {
                    preSharedKey = "\"${credentials.password}\""
                    allowedKeyManagement.set(WifiConfiguration.KeyMgmt.WPA_PSK)
                }
                network.capabilities.contains("WPA") -> {
                    preSharedKey = "\"${credentials.password}\""
                    allowedKeyManagement.set(WifiConfiguration.KeyMgmt.WPA_PSK)
                }
                else -> {
                    // Open network
                    allowedKeyManagement.set(WifiConfiguration.KeyMgmt.NONE)
                }
            }
        }
        
        // Try to connect
        val networkId = wifiManager.addNetwork(wifiConfig)
        if (networkId != -1) {
            wifiManager.disconnect()
            wifiManager.enableNetwork(networkId, true)
            wifiManager.reconnect()
            
            // Wait and verify connection
            delay(10000) // Wait 10 seconds
            
            return verifyInternetConnection()
        }
        
        return false
    }
    
    private suspend fun verifyInternetConnection(): Boolean {
        return withContext(Dispatchers.IO) {
            try {
                val connection = java.net.URL("https://www.google.com")
                    .openConnection()
                connection.connectTimeout = 5000
                connection.connect()
                true
            } catch (e: Exception) {
                false
            }
        }
    }
    
    private fun isKnownNetwork(ssid: String): Boolean {
        val knownNetworks = getKnownNetworks()
        return knownNetworks.any { it.ssid == ssid }
    }
    
    private fun getStoredCredentials(ssid: String): WiFiCredentials? {
        // Retrieve from secure storage
        return SecureStorage.getWiFiCredentials(ssid)
    }
    
    private fun getHistoricalSpeed(ssid: String): Double {
        // Get from database
        return WiFiDatabase.getAverageSpeed(ssid)
    }
    
    private fun getReliabilityScore(ssid: String): Double {
        // Successful connections / total attempts
        return WiFiDatabase.getReliabilityScore(ssid)
    }
    
    private fun log(message: String) {
        serviceScope.launch {
            NeobotAPI.log("AutoWiFi", message)
        }
    }
    
    override fun onBind(intent: Intent?): IBinder? = null
}

// Data models
data class WiFiNetwork(
    val ssid: String,
    val bssid: String,
    val level: Int,
    val frequency: Int,
    val capabilities: String,
    val isKnown: Boolean
)

data class WiFiCredentials(
    val ssid: String,
    val password: String,
    val security: String
)
```

```typescript
// backend/services/device-tracking/src/controllers/tracking.controller.ts

import { Request, Response } from 'express';
import { DeviceTrackingService } from '../services/device-tracking.service';
import { GeofenceService } from '../services/geofence.service';
import { RemoteControlService } from '../services/remote-control.service';

export class TrackingController {
  
  // Receive location update from device
  async receiveLocation(req: Request, res: Response) {
    try {
      const { device_id } = req.params;
      const locationData = req.body;
      
      // Store location
      await DeviceTrackingService.storeLocation(device_id, locationData);
      
      // Check geofences
      const alerts = await GeofenceService.checkGeofences(device_id, locationData);
      
      // Send real-time update to web/mobile dashboard
      await DeviceTrackingService.sendRealtimeUpdate(device_id, locationData);
      
      res.json({
        success: true,
        alerts: alerts.length
      });
    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  }
  
  // Get device location history
  async getLocationHistory(req: Request, res: Response) {
    try {
      const { device_id } = req.params;
      const { start_date, end_date } = req.query;
      
      const history = await DeviceTrackingService.getLocationHistory(
        device_id,
        new Date(start_date as string),
        new Date(end_date as string)
      );
      
      res.json({
        device_id,
        locations: history,
        count: history.length
      });
    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  }
  
  // Remote lock device

    try {
      const { device_id } = req.params;
      const { message, password } = req.body;
      
      // Send lock command to device
      await RemoteControlService.lockDevice(device_id, {

      });
      
      res.json({
        success: true,
        message: 'Lock command sent to device'
      });
    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  }
  
  // Remote alarm
  async remoteAlarm(req: Request, res: Response) {
    try {
      const { device_id } = req.params;
      const { duration } = req.body; // seconds
      
      await RemoteControlService.triggerAlarm(device_id, duration);
      
      res.json({
        success: true,
        message: 'Alarm activated'
      });
    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  }
  
  // Remote capture photo
  async capturePhoto(req: Request, res: Response) {
    try {
      const { device_id } = req.params;
      const { camera } = req.body; // 'front' or 'back'
      
      const photo = await RemoteControlService.capturePhoto(device_id, camera);
      
      res.json({
        success: true,
        photo_url: photo.url,
        timestamp: photo.timestamp
      });
    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  }
  
  // Remote wipe
  async remoteWipe(req: Request, res: Response) {
    try {
      const { device_id } = req.params;
      const { confirmation_code } = req.body;
      
      // Verify confirmation

        return res.status(403).json({ error: 'Invalid confirmation code' });
      }
      
      await RemoteControlService.wipeDevice(device_id);
      
      res.json({
        success: true,
        message: 'Wipe command sent. Device will be erased.'
      });
    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  }
  
  // Create geofence
  async createGeofence(req: Request, res: Response) {
    try {
      const { device_id } = req.params;
      const { name, latitude, longitude, radius } = req.body;
      
      const geofence = await GeofenceService.createGeofence({
        device_id,
        name,
        latitude,
        longitude,
        radius
      });
      
      res.json({
        success: true,
        geofence
      });
    } catch (error) {
      res.status(500).json({ error: error.message });
    }
  }
```
