## 🚀 QUICK START (5 MINUTES)

```kotlin
// Step 1: Install (2 minutes)
implementation 'com.neobot:object-recognition:7.0.0'

// Step 2: Initialize (30 seconds)
val neobot = AdvancedObjectRecognition()
neobot.initialize()

// Step 3: Single Recognition
val result = neobot.recognizeObject(
    image = cameraImage,
    language = "id", // or "en", "ja", "zh", etc
    voiceOutput = true
)

// Step 4: Access Results
for (obj in result.objects) {
    println("Found: ${obj.name}")
    println("Brand: ${obj.brand?.name}")
    println("Description: ${obj.description}")
    
    // Access recommendations
    for (rec in result.recommendations) {
        println("${rec.icon} ${rec.title}")
        println("   ${rec.description}")
    }
}

// Step 5: Continuous Recognition (Optional)
val realtime = RealTimeRecognition()
realtime.startContinuousRecognition(
    deviceType = DeviceType.SMARTPHONE,
    language = "id"
) { result ->
    // Called for each new object detected
    showNotification(result)
}

// Done! 🎉
```
